1. BUILD MONOREPO

- pnpm init
- pnpm add -D turbo typescript

2. CREATE A pnpm-workspace.yaml WITH CONTENT:

packages:
  - "apps/*"
  - "packages/*"

3. CREATE A turbo.json WITH CONTENT:

{
  "$schema": "https://turbo.build/schema.json",
  "tasks": {
    "build": {
      "dependsOn": ["^build"],
      "outputs": ["dist/**", ".next/**"]
    },
    "dev": {
      "cache": false
    },
    "lint": {},
    "test": {},
    "typecheck": {}
  }
}

4. IMPORT REPOS OF YOUR CHOOSEN ARCHITECTURE INTO THE /apps FOLDER 
               OR
   pnpm create next-app /apps/[app-name] FOR FRONTNED
   pnpm create nestjs /apps/[app-name] FOR BACKEND

5. pnpm add express cors dotenv
   pnpm add -D typescript tsx @types/node

6. ROOT package.json:

{
  "name": "repo",
  "private": true,
  "scripts": {
    "dev": "turbo run dev",
    "build": "turbo run build",
    "lint": "turbo run lint",
    "test": "turbo run test",
    "typecheck": "turbo run typecheck"
  }
}

TO DEPLOY BACKEND TO AWS

1. CREATE A Dockerfile IN BACKEND WITH CONTENT

FROM node:22-alpine

WORKDIR /app

RUN npm install -g pnpm@10.27.0

COPY . .

RUN pnpm install

RUN pnpm build

EXPOSE 3000

CMD ["pnpm", "start"]

2. RUN - "docker build -t backend ." in backend
3. RUN - "docker run -p 5001:3000 backend"
4. CHANGE THE BACKEND PORT TO 3000 IN .env
5. RUN "docker run -p 3000:3000 backend" AND IF IT IS RUN FINE, THEN MOVE TO AWS ERC.
